    const dia = String(hoje.getDate()).padStart(2, "0");

    // Enviar agendamento para WhatsApp
    function enviarWhatsapp() {

      // IMPORTANTE:
      // Coloque aqui o número real do WhatsApp.
      // Exemplo: 5577999999999
      const telefone = "5571999042120";

      const nome = document.getElementById("cliente").value.trim();
      const data = document.getElementById("data").value;
      const horario = document.getElementById("horario").value;

      const pagamentoSelecionado =
        document.querySelector('input[name="pagamento"]:checked');


      // Verificar nome
      if (!nome) {
        alert("Por favor, informe seu nome.");
        return;
      }


      // Verificar data
      if (!data) {
        alert("Por favor, escolha uma data.");
        return;
      }


      // Verificar horário
      if (!horario) {
        alert("Por favor, escolha um horário.");
        return;
      }


      // Verificar serviço
      const checkboxes =
        document.querySelectorAll('input[name="servico"]:checked');

      if (checkboxes.length === 0) {
        alert("Selecione pelo menos um serviço.");
        return;
      }


      // Verificar pagamento
      if (!pagamentoSelecionado) {
        alert("Escolha uma forma de pagamento.");
        return;
      }


      // Serviços e total
      let servicos = [];
      let total = 0;

      checkboxes.forEach(function(cb) {

        servicos.push(cb.value);

        total += parseFloat(
          cb.getAttribute("data-price")
        );

      });


      // Formatar data para DD/MM/AAAA
      const partesData = data.split("-");

      const dataFormatada =
        `${partesData[2]}/${partesData[1]}/${partesData[0]}`;


      // Criar mensagem
      const mensagem =
        `*✂️ NOVO AGENDAMENTO - BARBEARIA DO DAVID*%0A%0A` +

        `👤 *Cliente:* ${nome}%0A` +

        `💈 *Serviços:* ${servicos.join(", ")}%0A` +

        `💰 *Total:* R$ ${total.toFixed(2).replace(".", ",")}%0A` +

        `📅 *Data:* ${dataFormatada}%0A` +

        `🕐 *Horário:* ${horario}%0A` +

        `💳 *Pagamento:* ${pagamentoSelecionado.value}`;


      // Abrir WhatsApp
      window.open(
        `https://wa.me/${telefone}?text=${mensagem}`,
        "_blank"
      );

    }

  </script>
